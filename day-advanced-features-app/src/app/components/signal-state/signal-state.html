💻 State Service Example

// todo-state.service.ts
private todos = signal<Todo[]>([]);

// Computed signals (derived state)
activeTodos = computed(() => 
  this.todos().filter(t => !t.completed)
);

activeCount = computed(() => 
  this.activeTodos().length
);

// Actions
addTodo(text: string): void {
  const newTodo: Todo = {
    id: Date.now(),
    text,
    completed: false
  };
  
  this.todos.update(todos => 
    [...todos, newTodo]
  );
}

✨ Benefits of Signal-Based State

  • ✅ Simple - No boilerplate, built into Angular
  • ✅ Performant - Fine-grained reactivity
  • ✅ Type-safe - Full TypeScript support
  • ✅ Testable - Easy to test services
  • ✅ Composable - Computed signals for derived state
  • ✅ Automatic - Effects for side effects (auto-save to localStorage)
